import type { EmbeddingVersion } from @/api/embeddingsApi";
import { DataTable } from "@/components/shared/data-table";
import { listEmbeddings, deleteEmbeddings } from @/api/embeddingsApi";
import { useState, useEffect } from "react";
interface EmbeddingsTabProps {
  selectedDocIds: string[];
  projectId: string;
}
export function EmbeddingsTab({
  selectedDocIds = [],
  projectId,
}: EmbeddingsTabProps) {
  const [embeddings, setEmbeddings] = useState<EmbeddingVersion[]>([]);
  useEffect(() => {
    listEmbeddings(projectId).then(setEmbeddings);
  }, []);
  const handleDelete = async (versionIds: string[]) => {
    await deleteEmbeddings(projectId, versionIds);
    // Refresh data
    const updated = await listEmbeddings(projectId);
    setEmbeddings(updated);
  };
  return <DataTable data={embeddings} onDelete={handleDelete} />;
}

  version_id: string;
  version_name: string;
  project_id: string;
  collection_id: string;
  version_number: number;

  version_name: string;
  project_id: string;
  doc_id: string[];

  version_ids: string[];
}
// Create embeddings for selected documents
export async function createEmbedding(
  projectId: string,
  data: CreateEmbeddingRequest
): Promise<EmbeddingVersion> {
  const res = await api.post/v1/project/${projectId}/embeddings/, {
    version_name: data.version_name,
    doc_id: data.doc_id,
  });
  return res.data;
}

export interface Document {
  doc_id: string;

  file_type: string;

  updated_at: string | null;
}
export async function getDocument(
  projectId: string,
  docId: string
): Promise<Document> {
  const res = await api.get/v1/project/${projectId}/documents/${docId});
  return res.data;
}
